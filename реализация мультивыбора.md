  3. State + toggle (строки 102–113)                                                                                                                                                                                                                                                     
                                                                                                                                                                                                                                                                                         
  const [vehicleFilters, setVehicleFilters] = useState<Set<string>>(new Set(['all']));                                                                                                                                                                                                   
                                                                                                                                                                                                                                                                                         
  const toggleFilter = (key: string) => {                                                                                                                                                                                                                                                
    setVehicleFilters(prev => {                                                                                                                                                                                                                                                          
      if (key === 'all') return new Set(['all']);     // клик "Все" → сброс                                                                                                                                                                                                              
      const next = new Set(prev);
      next.delete('all');                              // любой другой клик убирает "Все"
      if (next.has(key)) next.delete(key); else next.add(key);  // toggle
      return next.size === 0 ? new Set(['all']) : next; // пусто → снова "Все"
    });
  };

  Это и есть ядро мультивыбора: Set<string>, клик по "Все" сбрасывает всё, клик по любому другому убирает "Все" и переключает сам себя.
